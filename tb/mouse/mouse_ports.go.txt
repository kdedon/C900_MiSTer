// Dumps the Go HR mouse (FN#7) port words after each injected move:
// one line per step, "dx dy left middle right x400 x402" (hex words).
// dx/dy are detents, right/down positive.
package main

import (
	"c900/chips/hrdisp"
	"fmt"

	"z8000/system"
	"z8000/zbus"
)

func pump(d *hrdisp.Display, n int) {
	for i := 0; i < n; i++ {
		d.TickBus(system.CardBus{PCLK: false, DS_n: true, RESET_n: true})
		d.TickBus(system.CardBus{PCLK: true, DS_n: true, RESET_n: true})
	}
}

func read(d *hrdisp.Display, port uint16) uint16 {
	out := d.TickBus(system.CardBus{CycleActive: true, AS_n: true, DSFalling: true, LatchedRW: true,
		LatchedStatus: zbus.StatusIO, LatchedAddr: port, PCLK: true, RESET_n: true})
	if !out.ADDrive {
		panic("port not driven")
	}
	return out.AD
}

func main() {
	type st struct {
		dx, dy  int
		l, m, r bool
	}
	steps := []st{
		{1, 0, false, false, false}, {0, 1, false, false, false}, {-1, 0, false, false, false},
		{0, -1, false, false, false}, {3, -2, true, false, false}, {-7, 5, false, true, false},
		{40, 40, false, false, true}, {-255, 0, true, true, true}, {0, -255, false, false, false},
		{300, -300, false, false, false}, {1030, 0, true, false, true}, {0, 1030, false, false, false},
		{-1030, -1030, false, true, false}, {511, 512, false, false, false}, {-2, 3, false, false, false},
	}
	d := hrdisp.New()
	pump(d, 10)
	for _, s := range steps {
		d.SetMouseButtons(s.l, s.r, s.m)
		d.InjectMouse(s.dx, s.dy)
		n := s.dx
		if n < 0 {
			n = -n
		}
		m := s.dy
		if m < 0 {
			m = -m
		}
		if m > n {
			n = m
		}
		pump(d, 5*n+20)
		b := func(v bool) int {
			if v {
				return 1
			}
			return 0
		}
		fmt.Printf("%d %d %d %d %d %04x %04x\n", s.dx, s.dy, b(s.l), b(s.m), b(s.r), read(d, 0x400), read(d, 0x402))
	}
}
